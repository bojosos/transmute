cmake_minimum_required(VERSION 3.20)
project(alchemy_cmake_deep C)
enable_testing()
add_library(util STATIC util.c)
add_executable(app main.c)
target_link_libraries(app PRIVATE util)
add_test(NAME app_runs COMMAND app 3)
set_tests_properties(app_runs PROPERTIES PASS_REGULAR_EXPRESSION "sum=14")
add_test(NAME app_fails COMMAND app)
set_tests_properties(app_fails PROPERTIES WILL_FAIL TRUE)
file(WRITE "${CMAKE_BINARY_DIR}/generated/config.txt" "configured\n")
